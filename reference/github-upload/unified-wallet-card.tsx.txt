import React,{useEffect,useState} from 'react';import type{RewardsGateway,RewardsSummary}from'../../lib/rewardsGateway';
export function UnifiedWalletCard({gateway,userId,pots}:{gateway:RewardsGateway;userId:string;pots:{id:string;name:string}[]}){const[s,setS]=useState<RewardsSummary>();const[dest,setDest]=useState('wallet');useEffect(()=>{gateway.getSummary(userId).then(setS)},[gateway,userId]);if(!s)return <div>Loading rewards…</div>;return <section className="rewards-wallet"><div><span>Zoryn Rewards</span><h3>{s.universalPoints.toLocaleString()} points</h3><p>€{(s.universalPoints/100).toFixed(2)} value · {s.pendingPoints} pending · {s.tier} tier</p></div><label>Send cashback to<select value={dest} onChange={async e=>{setDest(e.target.value);await gateway.setCashbackDestination(userId,e.target.value as any)}}><option value="wallet">Rewards wallet</option><option value="main">Main balance</option>{pots.map(p=><option key={p.id} value={`pot:${p.id}`}>{p.name} pot</option>)}</select></label></section>}
